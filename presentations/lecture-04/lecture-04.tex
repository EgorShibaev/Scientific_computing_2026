% 2
\begin{frame}{One hundred waiting times}
Record 100 waits. Compute their average.\\
Repeat with another 100 waits.
\vfill
\[
  T_i\sim\operatorname{Exp}(0.2),\qquad
  \mu=5\text{ min},\qquad \sigma=5\text{ min}.
\]
\vfill
How far might either average be from five minutes?
\takeaway{The observations follow the same model as Chapter 2.}
\end{frame}

% 3
\begin{frame}{Samples and sampling distributions}
\begin{definitionbox}{Sample mean and sampling distribution}
An i.i.d. sample has mutually independent observations with the same distribution.
Its sample mean is
\[
  \avgX=\frac{1}{n}\sum_{i=1}^{n}X_i.
\]
The distribution of $\avgX$ across repeated datasets is its
\textbf{sampling distribution}.
\end{definitionbox}
\vfill
Before collection: random variables $X_1,\ldots,X_n$.\\
After collection: recorded values $x_1,\ldots,x_n$.
\end{frame}

% 4
\begin{frame}{Standard deviation and standard error}
For i.i.d. observations with finite variance,
\[
  \Expect[\avgX]=\mu,\qquad \Var(\avgX)=\frac{\sigma^2}{n}.
\]
\begin{definitionbox}{Standard error}
A statistic's standard deviation across repeated samples.
For an i.i.d. sample mean,
\[
  \SE(\avgX)=\frac{\sigma}{\sqrt n}.
\]
\end{definitionbox}
One wait: $\mathrm{SD}=5$ minutes.\\
Average of 100 waits: $\mathrm{SE}=0.5$ minutes.
\takeaway{Individual waits keep their spread as the sample grows.}
\end{frame}

% 6
\begin{frame}{Unknown population variance}
\begin{definitionbox}{Sample variance}
\blockmath{
  S_n^2=\frac{1}{n-1}\sum_{i=1}^{n}(X_i-\avgX)^2,\qquad n\ge2.
}
\end{definitionbox}
The denominator accounts for centering at the sample mean.
We derive the correction in the estimation chapter.
\vfill
\[
  \widehat{\SE}(\avgX)=\frac{S_n}{\sqrt n},\qquad S_n=\sqrt{S_n^2}.
\]
\takeaway{The interpretation still depends on the sampling assumptions.}
\end{frame}

\begin{frame}{The probability of a large error}
Suppose the average wait should be within one minute of the population mean.
\vfill
\focusmath{\Prob\!\left(\left|\overline T_{100}-5\right|\ge1\right)=\,?}
\vfill
The standard error gives an error scale.\\
A probability statement needs more reasoning.
\vfill
\end{frame}

% 8
\begin{frame}{Markov's inequality}
\begin{theorembox}{Markov's inequality}
If $W\ge0$, $\Expect[W]<\infty$, and $a>0$, then
\[
  \Prob(W\ge a)\le\frac{\Expect[W]}{a}.
\]
\end{theorembox}
\textbf{Proof.} On the event $W\ge a$, the value of $W$ is at least $a$:
\[
  W\ge a\,\mathbf{1}_{\{W\ge a\}}
  \quad\Longrightarrow\quad
  \Expect[W]\ge a\,\Prob(W\ge a).
\]
\takeaway{Frequent large values force the mean upward.}
\end{frame}

% 9
\begin{frame}{Why Markov can be conservative}
Any nonnegative wait with mean five minutes:
\[
  \Prob(T\ge20)\le\frac{5}{20}=0.25.
\]
Under the exponential model:
\focusmath{\Prob(T\ge20)=e^{-4}\approx0.0183.}
\vfill
The bound is attainable: probability $\tfrac14$ at $20$ and $\tfrac34$ at $0$.
A mean alone cannot distinguish these distributions.
\end{frame}

% 10
\begin{frame}{Chebyshev's inequality}
\begin{theorembox}{Chebyshev's inequality}
If $\Expect[X]=\mu$ and $\Var(X)=\sigma^2<\infty$, then for $\varepsilon>0$,
\[
  \Prob(|X-\mu|\ge\varepsilon)\le\frac{\sigma^2}{\varepsilon^2}.
\]
\end{theorembox}
\textbf{Proof.} Apply Markov with $W=(X-\mu)^2$ and $a=\varepsilon^2$:
\[
  \Prob(|X-\mu|\ge\varepsilon)=\Prob(W\ge\varepsilon^2).
\]
\takeaway{Squaring deviations makes Markov apply to both tails.}
\end{frame}

% 11
\begin{frame}{A finite-sample guarantee}
\[
  \Prob(|\avgX-\mu|\ge\varepsilon)
  \le\min\!\left(1,\frac{\sigma^2}{n\varepsilon^2}\right).
\]
For the inbox, $\varepsilon=1$ minute and $n=100$:
\[
  \Prob(|\overline T_{100}-5|\ge1)\le\frac{25}{100}=0.25.
\]
To make the bound at most $\delta$, it suffices that
\focusmath{n\ge\frac{\sigma^2}{\delta\varepsilon^2}.
  \qquad \delta=0.05\text{ gives }n=500.}
\takeaway{Use the true variance or a known upper bound.
A sample-variance plug-in needs further justification.}
\end{frame}

\renewcommand{\conspectsection}{4.3}
% 12
\begin{frame}{The weak law of large numbers}
\begin{theorembox}{Weak LLN, finite-variance version}
For i.i.d. observations with mean $\mu$ and finite variance,\\
every fixed tolerance $\varepsilon>0$ satisfies
\[
  \Prob(|\avgX-\mu|>\varepsilon)\longrightarrow0
  \qquad\text{as }n\to\infty.
\]
\end{theorembox}
\textbf{Proof.} Chebyshev gives
\[
  \Prob(|\avgX-\mu|>\varepsilon)
  \le\frac{\sigma^2}{n\varepsilon^2}\longrightarrow0.
\]
\takeaway{This is convergence in probability.}
\end{frame}

% 14
\begin{frame}{The scope of the law of large numbers}
The fixed-tolerance error probability tends to zero across repeated datasets.
\vfill
\begin{theorembox}{LLN with a finite first moment}
For i.i.d. observations, $\Expect[|X_1|]<\infty$ is enough:
\[
  \avgX\xrightarrow{\Prob}\Expect[X_1].
\]
\end{theorembox}
Finite variance is enough for our short proof.
\vfill
Sampling bias remains: recording only short waits learns the mean
of that selected population.
\end{frame}

\renewcommand{\conspectsection}{4.4}
% 15
\begin{frame}{The shape of the sampling error}
The LLN makes the error small.\\
What distribution describes the remaining fluctuations?
\vfill
\[
  \avgX-\mu\qquad\text{has standard deviation}\qquad\frac{\sigma}{\sqrt n}.
\]
\vfill
The exponential example has an exact Gamma answer.\\
For a complicated measurement or loss, we need an approximation.
\vfill
\end{frame}

% 16
\begin{frame}{Standardized sampling error}
\begin{definitionbox}{Standardized sample mean}
For $\sigma>0$, measure the error in standard-error units:
\[
  Z_n=\frac{\avgX-\mu}{\sigma/\sqrt n}
  =\frac{\sqrt n\,(\avgX-\mu)}{\sigma}.
\]
\end{definitionbox}
\[
  \Expect[Z_n]=0,\qquad\Var(Z_n)=1.
\]
\vfill
Rescaling keeps the shrinking fluctuations visible.\\
The new claim concerns their distributional shape.
\end{frame}

% 17
\begin{frame}{The central limit theorem}
\begin{theorembox}{Classical CLT}
For i.i.d. observations with mean $\mu$ and $0<\sigma^2<\infty$,
\[
  \Prob(Z_n\le z)\longrightarrow\Phi(z)
  \qquad\text{as }n\to\infty,
\]
for every real $z$.
\end{theorembox}
This is convergence in distribution to the standard Gaussian.
\[
  \avgX\ \overset{\text{approx.}}{\sim}
  \mathcal{N}\!\left(\mu,\frac{\sigma^2}{n}\right).
\]
\takeaway{Gaussian observations give this law exactly for every $n$.
The original observations keep their own distribution.}
\end{frame}

% 20
\begin{frame}{One minute equals two standard errors}
\[
  n=100,\qquad\SE(\overline T_{100})=\frac{5}{\sqrt{100}}=0.5.
\]
\vfill
\[
  \Prob(|\overline T_{100}-5|>1)=\Prob(|Z_{100}|>2)
\]
\focusmath{\approx 2\,[1-\Phi(2)]\approx0.04550.}
\vfill
About $4.55\%$ under the Gaussian approximation.
\end{frame}

% 23
\begin{frame}{Finite-sample accuracy of the CLT}
There is no universal sample size, such as $30$, that guarantees
a useful Gaussian approximation.
\vfill
Skewness and rare extreme values can delay convergence.
\vfill
A small absolute CDF error can still mean a large relative error
for a tiny tail probability.
\end{frame}

\renewcommand{\conspectsection}{4.5}
% 24
\begin{frame}{A shared calibration error}
\[
  X_i=\mu+B+\epsilon_i.
\]
$B$ stays fixed within one dataset. A new independently calibrated
dataset gets a fresh $B$.
\vfill
The reading errors $\epsilon_i$ are centered and independent,
and independent of the centered calibration error $B$.
\focusmath{\avgX=\mu+B+\frac1n\sum_{i=1}^{n}\epsilon_i.}
\end{frame}

% 25
\begin{frame}{Covariance leaves a variance floor}
Each reading has variance $\sigma^2$. Distinct pairs have
correlation $\rho$, with $0\le\rho\le1$.
\[
  \begin{aligned}
    \Var(\avgX)
      &=\frac{n\sigma^2+n(n-1)\rho\sigma^2}{n^2}\\[0.35em]
      &=\sigma^2\!\left[\rho+\frac{1-\rho}{n}\right].
  \end{aligned}
\]
There are $n$ diagonal covariance terms and $n(n-1)$ off-diagonal terms.
\begin{resultbox}{Variance floor}
\blockmath{
  \Var(\avgX)\longrightarrow\rho\sigma^2
  \qquad\text{as }n\to\infty.
}
\end{resultbox}
\end{frame}

\begin{frame}{Finite mean and finite variance}
For i.i.d. Student observations with $\nu=\tfrac32$:
\focusmath{\Expect[|X_1|]<\infty,\qquad\Var(X_1)=\infty.}
The general LLN applies: the sample mean converges in probability to zero.
\vfill
The finite-variance Chebyshev argument and classical CLT do not apply.
There is no finite $\sigma$ for $\sigma/\sqrt n$.
\end{frame}

% 29
\begin{frame}{Monte Carlo estimation}
\begin{definitionbox}{Monte Carlo estimator}
For $I=\Expect[g(X)]$, draw $X_i$ independently from the target distribution:
\[
  \widehat I_n=\frac1n\sum_{i=1}^{n}g(X_i).
\]
\end{definitionbox}
$\Expect[|g(X)|]<\infty$ gives the LLN.\\
Finite positive $\tau^2$ also gives the CLT.
\[
  \SE(\widehat I_n)=\frac{\tau}{\sqrt n},\qquad\tau^2=\Var(g(X)).
\]
\takeaway{For unknown $\tau$: $\widehat{\SE}=S_Y/\sqrt n$, with $S_Y$
the sample SD of $g(X_i)$.}
\end{frame}

% 30
\begin{frame}{A Monte Carlo example with a known answer}
\[
  T\sim\operatorname{Exp}(0.2),\qquad g(t)=e^{-t}.
\]
Time is in minutes. The exponent uses one per minute.
\vfill
\[
  I=\Expect[g(T)]=\frac16,\qquad\Expect[g(T)^2]=\frac1{11}.
\]
\focusmath{\tau^2=\frac1{11}-\frac1{36}=\frac{25}{396}}
\takeaway{The known value lets us check the numerical method.}
\end{frame}

% 31
\begin{frame}{Precision has a computational cost}
\begin{definitionbox}{Root mean squared error}
\blockmath{
  \RMSE(\widehat I_n)
  =\sqrt{\Expect[(\widehat I_n-I)^2]}.
}
\end{definitionbox}
For our Monte Carlo average, $\RMSE(\widehat I_n)=\tau/\sqrt n$.
\par\smallskip
Four times as many evaluations halves RMSE.\\
A tenfold reduction costs one hundred times as many.
\[
  n=1000\qquad\Longrightarrow\qquad\RMSE\approx0.00795.
\]
\focusmath{\RMSE\le0.005\qquad\text{needs }n\ge2526.}
\takeaway{An RMSE target does not bound the error in every run.}
\end{frame}

% 33
\begin{frame}{Uncertainty in a test-set average}
Fix $f$ before evaluation. Use i.i.d. test observations independent
of all data used to choose $f$.
\begin{definitionbox}{Risk and empirical risk}
\blockmath{
  L_i=\ell(f(X_i),Y_i),\qquad R(f)=\Expect[L_i],
  \qquad\widehat R_n(f)=\frac1n\sum_{i=1}^{n}L_i.
}
\end{definitionbox}
Population accuracy $0.8$, with $400$ independent test examples:
\focusmath{\SE=\sqrt{\frac{0.8\cdot0.2}{400}}=0.02.}
\takeaway{Test-sample variation only. Choosing $f$ on this test set breaks the fixed-model premise.}
\end{frame}

% 34
\begin{frame}{Batch size and the variance of an average}
\begin{resultbox}{Variance of a batch average}
At a fixed model, average $b$ independent losses, each with variance $\tau_L^2$.
\[
  \Var(\text{batch average})=\frac{\tau_L^2}{b}.
\]
\end{resultbox}
A fourfold increase in $b$ halves the standard deviation.
\vfill
For a fixed dataset, sampling indices with replacement uses its empirical
loss variance. Dependence and sampling without replacement require
different variance calculations.
\end{frame}
