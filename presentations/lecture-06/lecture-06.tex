\begin{frame}{Five waiting times, one fitted rate}
\slidespacing
The observations from Chapter 5 were
\[
  1,\ 2,\ 3,\ 4,\ 5\text{ minutes}.
\]
For independent exponential waits with unknown rate $\lambda$,
\[
  \widehat\lambda=\frac n{\sum_{i=1}^n t_i}
  =\frac5{15}=\frac13\text{ per minute}.
\]
A new sample of five waits usually gives a different estimate.
We want an interval for the same unknown rate $\lambda$.
\end{frame}

% Slide 3
\begin{frame}{Sampling distribution of an estimate}
\slidespacing
Keep the population rate $\lambda$ fixed. Repeat the same experiment:
\[
  \begin{array}{ccc}
    \text{Sample 1: }(T_1^{(1)},\ldots,T_n^{(1)})
      &\longmapsto&\widehat\lambda^{(1)}=n/\sum_iT_i^{(1)}\\[0.8em]
    \text{Sample 2: }(T_1^{(2)},\ldots,T_n^{(2)})
      &\longmapsto&\widehat\lambda^{(2)}=n/\sum_iT_i^{(2)}\\[0.8em]
    \vdots&&\vdots
  \end{array}
\]
\begin{definitionbox}{Sampling distribution}
The distribution of an estimate across new samples of the same size
from the same population.
\end{definitionbox}
\end{frame}

\begin{frame}{Confidence intervals}
\slidespacing
Write $\theta$ for the unknown population parameter, such as $\mu$ or $\lambda$.
\begin{definitionbox}{Confidence interval with level $1-\alpha$}
A rule takes a sample $X=(X_1,\ldots,X_n)$ and returns $[L(X),R(X)]$,
with
\[
  \Prob_\theta\{L(X)\le\theta\le R(X)\}\ge1-\alpha
  \quad\text{for every allowed }\theta.
\]
Here $\Prob_\theta$ means sampling data with true parameter $\theta$.
For $95\%$ confidence, $\alpha=0.05$.
\end{definitionbox}
An \textbf{exact} interval has this guarantee at the stated sample size.
An \textbf{approximate} interval may have different actual coverage.
\end{frame}

\begin{frame}{Coverage belongs to the interval rule}
\slidespacing
Under the Gaussian model, before collecting data, the endpoints are random:
\[
  \Prob_\mu\!\left\{
  \bar X-1.96\frac{\sigma}{\sqrt n}\le\mu\le
  \bar X+1.96\frac{\sigma}{\sqrt n}\right\}\approx0.95.
\]
After observing $n=25$, $\bar x=10.4$, with known $\sigma=2$:
\[
  C(x)=[9.616,\ 11.184].
\]
Both this interval and $\mu$ are fixed. It either covers $\mu$ or misses it.
\medskip
The $95\%$ refers to the fraction of intervals covering $\mu$ across
repeated experiments. A posterior probability would require a prior.
\end{frame}

% Slide 8
\begin{frame}{Quantiles give the probability cutoffs}
\slidespacing
\begin{definitionbox}{Quantile for a continuous, increasing CDF}
The $p$-quantile $q_p$ is the cutoff satisfying
\[
  \Prob(Y\le q_p)=p.
\]
For the standard Gaussian $Z\sim\mathcal N(0,1)$, write $z_p$.
\end{definitionbox}
\[
  z_{0.025}\approx-1.96,\qquad z_{0.975}\approx1.96.
\]
\[
  \Prob(-1.96\le Z\le1.96)\approx0.95.
\]
Each excluded tail has probability $0.025$.
\end{frame}

% Slide 9
\begin{frame}{A Gaussian mean with known variance}
\slidespacing
Assume $X_1,\ldots,X_n$ are i.i.d. $\mathcal N(\mu,\sigma^2)$,
with known $\sigma>0$.
\[
  \bar X\sim\mathcal N\!\left(\mu,\frac{\sigma^2}n\right),
  \qquad
  Z=\frac{\bar X-\mu}{\sigma/\sqrt n}\sim\mathcal N(0,1).
\]
\begin{definitionbox}{Pivot}
A function of the sample and unknown parameters whose distribution is
known and does not depend on any unknown parameter.
\end{definitionbox}
Here $Z$ is a pivot: its distribution is $\mathcal N(0,1)$ for every $\mu$.
\end{frame}

% Slide 10
\begin{frame}{Rearranging the pivot gives an interval}
\slidespacing
The following inequalities describe the same event:
\[
  -z_{1-\alpha/2}\le\frac{\bar X-\mu}{\sigma/\sqrt n}
  \le z_{1-\alpha/2}
\]
\[
  \Longleftrightarrow\quad
  \bar X-z_{1-\alpha/2}\frac\sigma{\sqrt n}\le\mu\le
  \bar X+z_{1-\alpha/2}\frac\sigma{\sqrt n}.
\]
\begin{resultbox}{Exact Gaussian mean interval, known variance}
\[
  \bar X\ \pm\ z_{1-\alpha/2}\frac\sigma{\sqrt n}.
\]
\end{resultbox}
The half-width is $z_{1-\alpha/2}\sigma/\sqrt n$.
Multiplying $n$ by four halves it.
\end{frame}

% Conspect 6.3
% Slide 11
\begin{frame}{Estimating the unknown noise variance}
\slidespacing
For $n\ge2$, estimate $\sigma^2$ using
\[
  S^2=\frac1{n-1}\sum_{i=1}^n(X_i-\bar X)^2.
\]
Write $S=\sqrt{S^2}$.
The standardized error now has a random denominator:
\[
  \frac{\bar X-\mu}{S/\sqrt n}
  =\underbrace{\frac{\bar X-\mu}{\sigma/\sqrt n}}_{\mathcal N(0,1)}
   \underbrace{\frac\sigma S}_{\text{random}}.
\]
A sample with an unusually small $S$ makes this ratio larger in magnitude.
We need its distribution before choosing a cutoff.
\end{frame}

\begin{frame}{Chi-square is a Gamma distribution}
\slidespacing
\begin{definitionbox}{Chi-square with $\nu$ degrees of freedom}
For $\nu$ independent standard Gaussians,
\[
  Z_1,\ldots,Z_\nu\overset{\mathrm{iid}}\sim\mathcal N(0,1),
  \qquad \sum_{j=1}^{\nu}Z_j^2\sim\chi^2_\nu.
\]
\end{definitionbox}
Squaring one standard Gaussian gives, for $u>0$,
\[
  f_{Z^2}(u)=\frac{e^{-u/2}}{\sqrt{2\pi u}},
  \qquad Z^2\sim\operatorname{Gamma}\!\left(\tfrac12,\tfrac12\right).
\]
Independent Gamma variables with the same rate add their shapes:
\[
  \boxed{\chi^2_\nu=\operatorname{Gamma}\!\left(
    \text{shape}=\tfrac\nu2,\ \text{rate}=\tfrac12\right).}
\]
\end{frame}

% Second part of original slide 12.
% Source: https://online.stat.psu.edu/stat414/Lesson26, section 26.3.
% Slide 13
\begin{frame}{Why the sample variance gives $n-1$ degrees of freedom}
\slidespacing
For i.i.d. Gaussian data, the residuals have one constraint:
$\sum_i(X_i-\bar X)=0$.
\begin{resultbox}{Gaussian sample variance}
\[
  U=\frac{(n-1)S^2}{\sigma^2}
   =\sum_{i=1}^n\left(\frac{X_i-\bar X}{\sigma}\right)^2
   \sim\chi^2_{n-1}.
\]
$U$ is independent of $\bar X$.
\end{resultbox}
For $n=2$, this can be checked directly:
\[
  U=\left(\frac{X_1-X_2}{\sqrt2\,\sigma}\right)^2
  \sim\chi^2_1.
\]
The general proof uses a Gaussian coordinate rotation to separate the mean
and $n-1$ independent residual components. We omit this longer derivation.
\end{frame}

% Original slide 13.
% Slide 14
\begin{frame}{Student's distribution accounts for estimating $\sigma$}
\slidespacing
For independent $Z\sim\mathcal N(0,1)$ and $U\sim\chi^2_\nu$,
\[
  \frac Z{\sqrt{U/\nu}}\sim t_\nu.
\]
Insert $Z=\sqrt n(\bar X-\mu)/\sigma$ and $\nu=n-1$:
\[
  \frac Z{\sqrt{U/(n-1)}}
  =\frac{\sqrt n(\bar X-\mu)/\sigma}{S/\sigma}
  =\frac{\bar X-\mu}{S/\sqrt n}\sim t_{n-1}.
\]
\begin{resultbox}{Exact Gaussian mean interval, unknown variance}
\[
  \bar X\ \pm\ t_{n-1,\,1-\alpha/2}\frac S{\sqrt n}.
\]
$t_{\nu,p}$ denotes the $p$-quantile of $t_\nu$.
\end{resultbox}
\end{frame}

\begin{frame}{Non-Gaussian data: an approximate mean interval}
\slidespacing
Assume i.i.d. observations with $\Expect X_i=\mu$ and
$0<\Var(X_i)=\sigma^2<\infty$. The CLT and consistency of $S$ give
\[
  \frac{\sqrt n(\bar X-\mu)}S
  =\underbrace{\frac{\sqrt n(\bar X-\mu)}\sigma}
      _{\xrightarrow{d}\,\mathcal N(0,1)}
   \underbrace{\frac\sigma S}_{\xrightarrow{P}\,1}
  \xrightarrow{d}\mathcal N(0,1).
\]
\begin{resultbox}{Large-sample approximation}
\[
  \bar X\ \pm\ 1.96\frac S{\sqrt n}
  \qquad\text{targets }95\%\text{ coverage as }n\to\infty.
\]
\end{resultbox}
Gaussian data give an exact Student interval at every $n\ge2$.
For other distributions, skewness and heavy tails can slow the approximation.
\end{frame}

\begin{frame}{An exact pivot for the exponential rate}
\slidespacing
Assume $T_1,\ldots,T_n\overset{\mathrm{iid}}\sim\operatorname{Exp}(\lambda)$.
The sum rule from Chapter 2 gives
\[
  W=\sum_{i=1}^nT_i\sim\operatorname{Gamma}(n,\text{rate}=\lambda).
\]
Scaling changes the rate to $1/2$, so
\[
  2\lambda W\sim\operatorname{Gamma}(n,\text{rate}=1/2)=\chi^2_{2n}.
\]
Write $c_{\nu,p}$ for the $p$-quantile of $\chi^2_\nu$. Since $W>0$,
\[
  c_{2n,0.025}\le2\lambda W\le c_{2n,0.975}
  \quad\Longleftrightarrow\quad
  \frac{c_{2n,0.025}}{2W}\le\lambda\le\frac{c_{2n,0.975}}{2W}.
\]
The event on each side has probability $0.95$.
\end{frame}

\begin{frame}{Accuracy is a sample mean of Bernoulli outcomes}
\slidespacing
Fix a classifier with population accuracy $p$. Test it on $n$ i.i.d. examples.
Let $Y_i=1$ if prediction $i$ is correct, and $Y_i=0$ otherwise.
\[
  Y_i\sim\operatorname{Bernoulli}(p),\quad
  k=\sum_iY_i,\quad \widehat p=\bar Y=\frac kn,\quad
  \Var(\widehat p)=\frac{p(1-p)}n.
\]
\begin{resultbox}{Approximate 95\% Wald interval}
\[
  \widehat p\ \pm\ 1.96\sqrt{\frac{\widehat p(1-\widehat p)}n}.
\]
\end{resultbox}
\[
  k=84,\ n=100:\qquad0.84\pm1.96\sqrt{\frac{0.84\cdot0.16}{100}}
  \approx[0.768,\ 0.912].
\]
\end{frame}

% Original slide 20.
% Slide 19
\begin{frame}{Twenty correct predictions do not prove perfect accuracy}
\slidespacing
Suppose all $20$ test predictions are correct:
\[
  \widehat p=1,\qquad
  \sqrt{\frac{\widehat p(1-\widehat p)}{20}}=0,
  \qquad C_{\mathrm{Wald}}=[1,1].
\]
Yet a classifier with true accuracy $p=0.95$ gives this result with probability
\[
  \Prob(k=20\mid p=0.95)=0.95^{20}\approx0.358.
\]
The estimated variance collapses at $\widehat p=1$.
The same failure gives $[0,0]$ when $k=0$.
\end{frame}

\begin{frame}{Wilson's interval for a proportion}
\slidespacing
\begin{resultbox}{Wilson interval, nominal confidence $1-\alpha$}
With $\widehat p=k/n$ and $z=z_{1-\alpha/2}$,
\[
  \frac{\displaystyle\widehat p+\frac{z^2}{2n}
    \ \pm\ z\sqrt{\frac{\widehat p(1-\widehat p)}n+
                      \frac{z^2}{4n^2}}}
       {\displaystyle1+\frac{z^2}n}.
\]
\end{resultbox}
It avoids Wald's zero-width boundary intervals and usually improves coverage
near $0$ and $1$. Its coverage is still approximate.
\end{frame}

\begin{frame}{The empirical distribution}
\slidespacing
We have one sample, $x=(1,2,3,4,5)$, but want to imitate new samples.
\begin{definitionbox}{Empirical distribution}
Assign probability $1/n$ to each of the $n$ recorded observations.
Repeated values receive their combined probability.
\end{definitionbox}
For the five observed waits, a draw $X^*$ has probabilities
\[
  \begin{array}{c|ccccc}
    \text{value}&1&2&3&4&5\\\hline
    \text{probability}&1/5&1/5&1/5&1/5&1/5
  \end{array}
\]
The star marks a draw from the observed sample, rather than from the unknown
population. The recorded values stay fixed during resampling.
\end{frame}

% Original slide 25.
% Slide 23
\begin{frame}{Sampling with replacement}
\slidespacing
Select $n=5$ recorded observations independently, allowing repetition:
\[
  x=(1,2,3,4,5),\qquad\bar x=3.
\]
One possible bootstrap sample is
\[
  x^*=(1,1,2,4,5),\qquad\bar x^*=\frac{13}5=2.6.
\]
\begin{definitionbox}{Bootstrap sample}
$n$ independent draws from the empirical distribution, with replacement.
The resample has the same size as the original sample.
\end{definitionbox}
Without replacement, taking all five observations only changes their order:
the mean would always stay $3$.
\end{frame}

\begin{frame}{Which errors does the bootstrap imitate?}
\slidespacing
For a population mean $\mu$ and the one observed mean $\bar x$,
\par\medskip
{\renewcommand{\arraystretch}{1.7}
\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}lll@{}}
\toprule
 & Fresh population sample & Bootstrap resample\\
\midrule
Source mean & $\mu$ (unknown) & $\bar x$ (known)\\
Sample estimate & $\bar X$ & $m_b$\\
Error & $\bar X-\mu$ & $m_b-\bar x$\\
\bottomrule
\end{tabular*}}
\par\medskip
Use the resampling errors $m_b-\bar x$ to approximate the real error
$\bar X-\mu$. For i.i.d. data with finite, positive variance, this works
as $n$ grows. Small samples can give a poor approximation.
\end{frame}

% Slide 26
\begin{frame}{Bootstrap standard error}
\slidespacing
The $B$ resample means are $m_1,\ldots,m_B$.
Their average is $\bar m=B^{-1}\sum_{b=1}^B m_b$.
\begin{definitionbox}{Bootstrap standard error of the sample mean}
The sample standard deviation of the resample means:
\[
  \widehat{\operatorname{SE}}_{\mathrm{boot}}
  =\sqrt{\frac1{B-1}\sum_{b=1}^B(m_b-\bar m)^2}.
\]
\end{definitionbox}
Five waits $(1,2,3,4,5)$: $\bar x=3$ and empirical variance $2$.
\[
  \Var(\bar X^*)=\frac25,\qquad
  \widehat{\operatorname{SE}}_{\mathrm{boot}}\approx\sqrt{\frac25}\approx0.632.
\]
\end{frame}

% Original slide 29, with the numerical example introduced here.
% Slide 27
\begin{frame}{Percentile bootstrap interval}
\slidespacing
Sort the $B$ resample means $m_1,\ldots,m_B$.
Write $q_p^*$ for their $p$-quantile.
\begin{definitionbox}{Percentile interval for the population mean $\mu$}
\[
  [q_{0.025}^*,\ q_{0.975}^*]\qquad\text{for nominal }95\%\text{ confidence}.
\]
\end{definitionbox}
A separate example uses $40$ observed waits, with mean $\bar x=4.495$ minutes.
Its $20{,}000$ resample means give
\[
  q_{0.025}^*=3.399,\quad q_{0.975}^*=5.682,
  \qquad C_{\mathrm{percentile}}=[3.399,\ 5.682].
\]
The interval takes the central $95\%$ of the bootstrap estimates.
Its population coverage is approximate.
\end{frame}

% Original slide 30. Derivation and explicit comparison.
% Source: https://web.stanford.edu/class/stats200/Lecture19.pdf
% Slide 28
\begin{frame}{Basic bootstrap interval: subtract the estimated errors}
\slidespacing
The errors $m_b-\bar x$ have approximate central bounds
\[
  [q_{0.025}^*-\bar x,\ q_{0.975}^*-\bar x]
  \approx[\,{-1.096},\ 1.187\,].
\]
Use these as approximate bounds on the real error:
\[
  -1.096\le\bar x-\mu\le1.187
  \quad\Longleftrightarrow\quad
  \bar x-1.187\le\mu\le\bar x+1.096.
\]
\begin{definitionbox}{Basic bootstrap interval}
\[
  [\,2\bar x-q_{0.975}^*,\ 2\bar x-q_{0.025}^*\,]
  \approx[3.31,\ 5.59].
\]
\end{definitionbox}
An upward estimation error means the estimate is too high, so we subtract it.
\end{frame}

\begin{frame}{Two classifiers on the same test examples}
\slidespacing
Two fixed classifiers predict the same $n=200$ independent test examples.
\par\smallskip
{\renewcommand{\arraystretch}{1.4}
\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}lrr@{}}
\toprule
 & B correct & B incorrect\\
\midrule
A correct & 150 & 30\\
A incorrect & 10 & 10\\
\bottomrule
\end{tabular*}}
\[
  \widehat p_A=\frac{150+30}{200}=0.90,\qquad
  \widehat p_B=\frac{150+10}{200}=0.80.
\]
\begin{definitionbox}{Population difference and its estimate}
$p_A,p_B$ are the population accuracies. Their difference is $\Delta$:
\[
  \Delta=p_A-p_B,\qquad
  \widehat\Delta=\widehat p_A-\widehat p_B=0.10.
\]
\end{definitionbox}
\end{frame}

% Original slide 33, first equality explained.
% Slide 32
\begin{frame}{The accuracy difference is a mean of paired differences}
\slidespacing
Define $A_i=1$ if A is correct on example $i$, and $0$ otherwise.
Define $B_i$ in the same way. Then set $d_i=A_i-B_i$:
\[
  d_i=\begin{cases}
    +1 & \text{A correct, B incorrect: 30 examples},\\
    -1 & \text{A incorrect, B correct: 10 examples},\\
     0 & \text{both correct or both incorrect: 160 examples}.
  \end{cases}
\]
\[
  \bar d=\frac1n\sum_i(A_i-B_i)
  =\frac1n\sum_iA_i-\frac1n\sum_iB_i
  =\widehat p_A-\widehat p_B=\widehat\Delta.
\]
\[
  \bar d=\frac{30-10}{200}=0.10.
\]
Both agreements give $d_i=0$: $1-1=0$ and $0-0=0$.
\end{frame}

% Original slide 33, standard-error equality explained.
% Slide 33
\begin{frame}{Computing the standard error of the difference}
\slidespacing
Use the sample-mean formula, now for the $200$ observed $d_i$ values:
\[
  s_d^2=\frac1{n-1}\sum_i(d_i-\bar d)^2,
  \qquad\widehat{\operatorname{SE}}(\widehat\Delta)=\frac{s_d}{\sqrt n}.
\]
Since $d_i^2=1$ on $40$ examples and $0$ on the other $160$,
\[
  \frac1n\sum_i d_i^2=\frac{40}{200}=0.20,
  \qquad \bar d=0.10.
\]
Expand the squared deviations: $\sum_i(d_i-\bar d)^2=\sum_i d_i^2-n\bar d^2$.
\[
  \frac{s_d}{\sqrt n}
  =\sqrt{\frac{\frac1n\sum_i d_i^2-\bar d^2}{n-1}}
  =\sqrt{\frac{0.20-0.10^2}{199}}\approx0.0309.
\]
The approximate $95\%$ interval is $0.10\pm1.96(0.0309)=[0.039,0.161]$.
\end{frame}

% Original slide 34, expressed using already-defined paired outcomes.
% Slide 34
\begin{frame}{Paired bootstrap: resample the two outcomes together}
\slidespacing
The observed rows are $(A_i,B_i)$, for $i=1,\ldots,n$.
Draw $n$ indices $I_1,\ldots,I_n$ independently and uniformly from
$\{1,\ldots,n\}$, with replacement.
\[
  \text{Resampled row }j=(A_{I_j},B_{I_j}),\qquad
  d_j^*=A_{I_j}-B_{I_j}.
\]
Write $\widehat p_A^*,\widehat p_B^*$ for accuracies on this resample.
\begin{resultbox}{One bootstrap accuracy difference}
\[
  \widehat\Delta^*=\frac1n\sum_{j=1}^n d_j^*
  =\widehat p_A^*-\widehat p_B^*.
\]
\end{resultbox}
Use the same indices for A and B. For example,
$(A_i,B_i)=(1,1)$ must remain an agreement in the resample.
\end{frame}

\begin{frame}{More resamples use the same observations}
\slidespacing
Keep $x=(1,2,3,4,5)$ fixed, so $n=5$.
Every bootstrap sample still has five observations. Increase only $B$,

\begin{frame}{More observations reduce uncertainty in the mean}
\slidespacing
For comparison, assume independent observations have population variance
$\sigma^2=2$. Then
\[
  \operatorname{SE}(\bar X)=\frac{\sigma}{\sqrt n}=\sqrt{\frac2n}.
\]
\par\medskip
{\renewcommand{\arraystretch}{1.5}
\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}rr@{}}
\toprule
Independent observations $n$ & Standard error of the sample mean\\
\midrule
$5$ & $0.632$\\
$20$ & $0.316$\\
$80$ & $0.158$\\
\bottomrule
\end{tabular*}}
\par\medskip
Four times as many independent observations halve the standard error.
Increasing $B$ improves the bootstrap calculation; increasing $n$ provides
new observations.
\end{frame}
